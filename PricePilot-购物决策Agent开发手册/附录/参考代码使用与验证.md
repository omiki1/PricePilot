## 参考代码：统一演示报价工厂（模块参考）

目标 `app/test/fixtures.py`；仅依赖统一 Schema。example.com 是保留示例域名，以下不代表 Amazon 或京东真实商品，不应作为可购买链接发布。

```python
from datetime import datetime, timedelta
from decimal import Decimal
from app.ai.agent.multi_agent.schema.shopping_schema import Offer, ProductIdentity


def make_offer(now: datetime, item_price: str = "2999.00", **changes) -> Offer:
    identity_values = {
        "brand": "DemoAudio", "model": "Commute One", "generation": "1",
        "variant": "standard", "storage": "not_applicable", "color": "black",
        "region": "CN", "condition": "new", "bundle": "standard",
        "warranty": "CN-1year",
    }
    identity = ProductIdentity(
        **identity_values,
        field_evidence={key: ["demo-evidence"] for key in identity_values},
    )
    data = {
        "offer_id": "demo-jd-1", "product_id": "demo-product-a",
        "source_product_id": "fixture-sku-1", "platform": "jd",
        "marketplace": "demo-cn", "title": "演示通勤耳机A",
        "identity": identity, "product_url": "https://example.com/demo/item-a",
        "currency": "CNY", "destination": "CN-demo",
        "buyer_context_hash": "public-demo",
        "item_price": Decimal(item_price), "shipping": Decimal("0"),
        "tax": Decimal("0"), "stock_status": "in_stock", "eligibility": "eligible",
        "observed_at": now, "valid_until": now+timedelta(minutes=15),
        "evidence_ids": ["demo-evidence"], "data_mode": "demo",
    }
    data.update(changes)
    return Offer.model_validate(data)
```

别把演示工厂用于真实来源映射。默认零运税只表示该虚构样本明确包邮且无额外税费，真实来源未知时必须填None。
